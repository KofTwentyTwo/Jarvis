    /// Constructs and starts the memory subsystem.
    ///
    /// Bootstrap order (mirrors installVoice's six-step pattern):
    ///   1. Build deps: jarvis.db URL under Application Support.
    ///   2. Construct MemoryStore — opens DB + loads vec0.dylib + migrates.
    ///      Failure is non-fatal: agent runs without memory until the user
    ///      resolves it (vec0.dylib bundled by ops/codesign.sh).
    ///   3. Construct EmbeddingClient (loopback-only, MEM-04 sandbox).
    ///   4. Construct MemoryExtractor on top of OllamaProvider configured for
    ///      qwen2.5-coder:32b.
    ///   5. Construct MemoryExtractionOrchestrator (bounded AsyncChannel
    ///      capacity 32, dropOldest, serial drain). start() spawns the
    ///      consumer Task.
    ///   6. Construct MemoryExtractionCoordinator and subscribe it to
    ///      AgentOrchestrator.events; .turnEnd(.endTurn) -> enqueue.
    ///   7. Log success.
    @MainActor
    private func installMemory() async {
        // 1. DB URL.
        let dbURL = configFileURL()
            .deletingLastPathComponent()
            .appendingPathComponent("jarvis.db")

        // 2. MemoryStore.
        let store: MemoryStore
        do {
            store = try MemoryStore(databaseURL: dbURL)
        } catch {
            systemLogger?.warning(
                "installMemory: store init failed (vec0.dylib missing or DB locked?): \(String(describing: error))"
            )
            return
        }
        memoryStore = store

        // 3. EmbeddingClient — loopback-only enforced at construction (MEM-04).
        let embeddingClient: OllamaEmbeddingClient
        do {
            embeddingClient = try OllamaEmbeddingClient(
                baseURL: URL(string: "http://127.0.0.1:11434")!
            )
        } catch {
            systemLogger?.warning(
                "installMemory: embedding client init failed: \(String(describing: error))"
            )
            return
        }

        // 4. MemoryExtractor on OllamaProvider(qwen2.5-coder:32b).
        let extractorProvider = OllamaProvider(
            baseURL: URL(string: "http://127.0.0.1:11434")!
        )
        let extractor = MemoryExtractor(provider: extractorProvider, store: store)

        // 5. Background orchestrator. Held strongly; start() spawns drain.
        let memoryReplaySink = AppDelegateMemoryReplaySink(
            channel: orchToReplayChannel
        )
        let memoryOrch = MemoryExtractionOrchestrator(
            extractor: extractor,
            store: store,
            replaySink: memoryReplaySink,
            embedder: embeddingClient
        )
        await memoryOrch.start()
        memoryExtractionOrchestrator = memoryOrch

        // 6. Coordinator wiring — subscribe to AgentOrchestrator.events.
        //    AgentOrchestrator is owned by Phase 4 wiring (a later AppDelegate
        //    extension we read at task-time). If it's not yet present in this
        //    AppDelegate, log + skip; the coordinator can be wired later.
        guard let orchestratorEvents = self.agentOrchestratorEvents() else {
            systemLogger?.warning(
                "installMemory: AgentOrchestrator.events unavailable — extraction coordinator not started"
            )
            return
        }
        let coord = MemoryExtractionCoordinator(memoryOrchestrator: memoryOrch)
        memoryExtractionCoordinator = coord
        await coord.start(orchestratorEvents: orchestratorEvents)

        systemLogger?.info("installMemory: MemoryExtractionOrchestrator started")
    }

    /// Returns the live AgentOrchestrator's event stream if Phase 4 wiring is

    /// installMemory body doesn't depend on the AgentOrchestrator's exact

#!/usr/bin/env bash
# VISION-03 architectural-boundary guard.
#
# 1. packages/Vision/Package.swift must NOT depend on Voice or AgentOrchestrator.
# 2. No Swift source under packages/Vision/Sources/ may import Voice or
#    AgentOrchestrator, nor reference TTSEngine / TTSEngineActor /
#    AgentOrchestrator.runTurn / cancelAndSubmit.
# 3. Any file in App/ that holds a reference to PresenceSignalBus.stream
#    must not, on the same line or in the same function, reference TTSEngine
#    or AgentOrchestrator submit paths.
#
# Layer 3 is the loosest — a future refactor in App/ that subscribes to
# presence and then forwards into TTS is the catastrophic VISION-03 failure
# mode. The guard is intentionally over-permissive: any forbidden-token hit
# in a file containing PresenceSignalBus reference fails the script.
set -euo pipefail

REPO_ROOT="$(cd "$(dirname "${BASH_SOURCE[0]}")/.." && pwd)"
cd "$REPO_ROOT"

# Layer 1: Package.swift dependencies.
if grep -E 'package\(path: "\.\./Voice"\)|package\(path: "\.\./AgentCore"\).*AgentOrchestrator' \
        packages/Vision/Package.swift 2>/dev/null; then
    echo "VISION-03 violation (Layer 1): Vision/Package.swift depends on Voice or AgentOrchestrator." >&2
    exit 1
fi

# Layer 2: forbidden imports / type references inside Vision sources.
LAYER2_HITS=$(grep -rnE '^\s*import\s+(Voice|AgentOrchestrator)\b|TTSEngine|TTSEngineActor|AgentOrchestrator\.runTurn|cancelAndSubmit' \
    packages/Vision/Sources/ --include='*.swift' 2>/dev/null \
    | grep -vE '^[^:]*:[[:space:]]*//' || true)
if [[ -n "$LAYER2_HITS" ]]; then
    echo "VISION-03 violation (Layer 2): forbidden import or type reference in Vision sources:" >&2
    echo "$LAYER2_HITS" >&2
    exit 1
fi

# Layer 3: in any App/ file that holds a PresenceSignalBus reference, no
# co-located forbidden-token hit (per file).
APP_PRESENCE_FILES=$(grep -rl 'PresenceSignalBus\b' App/ --include='*.swift' 2>/dev/null || true)
for f in $APP_PRESENCE_FILES; do
    if grep -nE 'TTSEngine|TTSEngineActor|cancelAndSubmit|AgentOrchestrator\.runTurn' "$f" \
            | grep -vE '^[^:]*:[[:space:]]*//' >/dev/null 2>&1; then
        echo "VISION-03 violation (Layer 3): file references both PresenceSignalBus AND a forbidden TTS/runTurn token: $f" >&2
        grep -nE 'PresenceSignalBus|TTSEngine|TTSEngineActor|cancelAndSubmit|AgentOrchestrator\.runTurn' "$f" >&2

    fi

    private static func repoRoot() throws -> URL {
        var dir = URL(fileURLWithPath: #file).deletingLastPathComponent()
        for _ in 0..<10 {

            let parentPkg = dir.deletingLastPathComponent().appendingPathComponent("Package.swift")

            throw NSError(domain: "PhaseSevenGrepGateTests", code: 2,

        let errPipe = Pipe()
        task.standardOutput = outPipe
        task.standardError = errPipe
        try task.run()
        task.waitUntilExit()
        return task.terminationStatus
